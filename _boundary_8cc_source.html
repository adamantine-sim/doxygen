12 Boundary::Boundary(boost::property_tree::ptree const &database,
13  std::vector<dealii::types::boundary_id> const &boundary_ids)
14 {
15  // Check that the boundary ids provided by the user match the ones from the
16  // domain.
17  for (auto const &child_pair : database)
18  {
19  std::string main_string = child_pair.first;
20  std::string substring = "boundary_";
21  size_t pos = main_string.find(substring);
22  // Check if the substring was found
23  if (pos != std::string::npos)
24  {
25  main_string.erase(pos, substring.length());
26  auto it = std::find(boundary_ids.begin(), boundary_ids.end(),
27  std::stoi(main_string));
28  ASSERT_THROW(it != boundary_ids.end(),
29  "Provided boundary id " + main_string + " is not valid.");
30  }
31  }
32 
33  _types.resize(*std::max_element(boundary_ids.begin(), boundary_ids.end()) + 1,
34  BoundaryType::invalid);
35  // Set the default boundary condition. It can be overwritten for given
36  // boundary ids.
37  // PropertyTreeInput boundary.type
38  auto default_boundary_type_str = database.get_optional<std::string>("type");
39  if (default_boundary_type_str)
40  {
41  BoundaryType default_boundary =
42  parse_boundary_line(*default_boundary_type_str);
43  for (auto &boundary : _types)
44  {
45  boundary = default_boundary;
46  }
47  }
48 
49  // Check if the user wants to use different boundary conditions for different
50  // part of the domain's boundary.
51  for (auto const id : boundary_ids)
52  {
53  // PropertyTreeInput boundary.boundary_id.type
54  auto boundary_type_str = database.get_optional<std::string>(
55  "boundary_" + std::to_string(id) + ".type");
56  if (boundary_type_str)
57  {
58  _types[id] = parse_boundary_line(*boundary_type_str);
59  }
60  }
61 
62  // Check if the user wants to use different for the printed surface
63  // PropertyTreeInput boundary.printed_surface.type
64  auto surface_boundary_type_str =
65  database.get_optional<std::string>("printed_surface.type");
66  if (surface_boundary_type_str)
67  {
68  _types.back() = parse_boundary_line(*surface_boundary_type_str);
69  }
70 }

87 BoundaryType Boundary::parse_boundary_line(std::string boundary_type_str)
88 {
89  BoundaryType boundary_type = BoundaryType::invalid;
90  std::string delimiter = ",";
91 
92  std::unordered_map<std::string, BoundaryType> const bc_map{
93  {"adiabatic", BoundaryType::adiabatic},
94  {"radiative", BoundaryType::radiative},
95  {"convective", BoundaryType::convective},
96  {"clamped", BoundaryType::clamped},
97  {"traction_free", BoundaryType::traction_free}};
98 
99  size_t pos_str = 0;
100  while ((pos_str = boundary_type_str.find(delimiter)) != std::string::npos)
101  {
102  std::string boundary = boundary_type_str.substr(0, pos_str);
103  boundary_type |= bc_map.find(boundary)->second;
104  boundary_type_str.erase(0, pos_str + delimiter.length());
105  }
106  boundary_type |= bc_map.find(boundary_type_str)->second;
107 
108  return boundary_type;
109 }
