96 void Quaternion::pow(double const exp)
97 {
98  ASSERT(_is_valid, "");
99 
100  double const partial_norm = std::sqrt(_quaternion[1] * _quaternion[1] +
101  _quaternion[2] * _quaternion[2] +
102  _quaternion[3] * _quaternion[3]);
103  std::array<double, 3> const unit_vector = {{_quaternion[1] / partial_norm,
104  _quaternion[2] / partial_norm,
105  _quaternion[3] / partial_norm}};
106  double theta = std::atan2(partial_norm, _quaternion[0]);
107 
108  _quaternion[0] = std::cos(exp * theta);
109  _quaternion[1] = unit_vector[0] * std::sin(exp * theta);
110  _quaternion[2] = unit_vector[1] * std::sin(exp * theta);
111  _quaternion[3] = unit_vector[2] * std::sin(exp * theta);
112 
113  build_rotation_matrices();
114 }

116 void Quaternion::build_rotation_matrices()
117 {
118  double r = _quaternion[0];
119  double i = _quaternion[1];
120  double j = _quaternion[2];
121  double k = _quaternion[3];
122 
123  // Build the rotation matrix
124  _rotation_matrix[0][0] = 1.0 - 2.0 * (j * j + k * k);
125  _rotation_matrix[0][1] = 2.0 * (i * j - k * r);
126  _rotation_matrix[0][2] = 2.0 * (i * k + j * r);
127  _rotation_matrix[1][0] = 2.0 * (i * j + k * r);
128  _rotation_matrix[1][1] = 1.0 - 2.0 * (i * i + k * k);
129  _rotation_matrix[1][2] = 2.0 * (j * k - i * r);
130  _rotation_matrix[2][0] = 2.0 * (i * k - j * r);
131  _rotation_matrix[2][1] = 2.0 * (j * k + i * r);
132  _rotation_matrix[2][2] = 1.0 - 2.0 * (i * i + j * j);
133 
134  // Build the vectorized rotation matrix
135  for (int i = 0; i < 3; ++i)
136  {
137  for (int j = 0; j < 3; ++j)
138  {
139  _vec_rotation_matrix[i][j] = _rotation_matrix[i][j];
140  }
141  }
142 
143  // Build the inverse rotation matrix. The inverse of a unit quaternion is
144  // obtained by changing the sign of its imaginay components, i.e., the inverse
145  // of (r,i,j,k) is (r,-i,-j,-k).
146  _inv_rotation_matrix[0][0] = 1.0 - 2.0 * (j * j + k * k);
147  _inv_rotation_matrix[0][1] = 2.0 * (i * j + k * r);
148  _inv_rotation_matrix[0][2] = 2.0 * (i * k - j * r);
149  _inv_rotation_matrix[1][0] = 2.0 * (i * j - k * r);
150  _inv_rotation_matrix[1][1] = 1.0 - 2.0 * (i * i + k * k);
151  _inv_rotation_matrix[1][2] = 2.0 * (j * k + i * r);
152  _inv_rotation_matrix[2][0] = 2.0 * (i * k + j * r);
153  _inv_rotation_matrix[2][1] = 2.0 * (j * k - i * r);
154  _inv_rotation_matrix[2][2] = 1.0 - 2.0 * (i * i + j * j);
155 
156  // Build the vectorized inverse rotation matrix
157  for (int i = 0; i < 3; ++i)
158  {
159  for (int j = 0; j < 3; ++j)
160  {
161  _vec_inv_rotation_matrix[i][j] = _inv_rotation_matrix[i][j];
162  }
163  }
164 }
