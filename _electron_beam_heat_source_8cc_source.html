21 template <int dim>
22 void ElectronBeamHeatSource<dim>::update_time(double time)
23 {
24  dealii::Point<3> const &path = this->_scan_path.value(time);
25  // Copy the scalar value of path to the vectorized data type
26  for (unsigned int d = 0; d < 3; ++d)
27  {
28  _beam_center(d) = path(d);
29  }
30  double segment_power_modifier = this->_scan_path.get_power_modifier(time);
31  _alpha =
32  -this->_beam.absorption_efficiency * this->_beam.max_power *
33  segment_power_modifier * _log_01 /
34  (dealii::numbers::PI * this->_beam.radius_squared * this->_beam.depth);
35  _depth = this->_beam.depth;
36  _radius_squared = this->_beam.radius_squared;
37 }

39 template <int dim>
40 double ElectronBeamHeatSource<dim>::value(dealii::Point<dim> const &point,
41  double const height) const
42 {
43  double const z = point[axis<dim>::z] - height;
44  if ((z + this->_beam.depth) < 0.)
45  {
46  return 0.;
47  }
48  else
49  {
50  double const distribution_z = -3. * std::pow(z / this->_beam.depth, 2) -
51  2. * (z / this->_beam.depth) + 1.;
52 
53  double xpy_squared =
54  std::pow(point[axis<dim>::x] - _beam_center[axis<dim>::x][0], 2);
55  if constexpr (dim == 3)
56  {
57  xpy_squared +=
58  std::pow(point[axis<dim>::y] - _beam_center[axis<dim>::y][0], 2);
59  }
60 
61  // Evaluating the exponential is very expensive. Return early if we know
62  // that the heat source will be small.
63  if (xpy_squared > 5. * this->_beam.radius_squared)
64  {
65  return 0.;
66  }
67 
68  // Electron beam heat source equation
69  double heat_source =
70  _alpha[0] *
71  std::exp(_log_01 * xpy_squared / this->_beam.radius_squared) *
72  distribution_z;
73 
74  return heat_source;
75  }
76 }

78 template <int dim>
79 dealii::VectorizedArray<double> ElectronBeamHeatSource<dim>::value(
80  dealii::Point<dim, dealii::VectorizedArray<double>> const &points,
81  dealii::VectorizedArray<double> const &height) const
82 {
83  auto const z = points[axis<dim>::z] - height;
84  auto const z_depth = z + _depth;
85  dealii::VectorizedArray<double> depth_mask;
86  for (unsigned int i = 0; i < depth_mask.size(); ++i)
87  {
88  depth_mask[i] = z_depth[i] < 0. ? 0. : 1.;
89  }
90  // If all the lanes of depth_mask are zero, we can return early
91  if (depth_mask.sum() == 0)
92  {
93  return depth_mask;
94  }
95 
96  auto xpy_squared = points[axis<dim>::x] - _beam_center[axis<dim>::x];
97  xpy_squared *= xpy_squared;
98  if constexpr (dim == 3)
99  {
100  auto y_squared = points[axis<dim>::y] - _beam_center[axis<dim>::y];
101  y_squared *= y_squared;
102  xpy_squared += y_squared;
103  }
104 
105  // Evaluating the exponential is very expensive. Return early if we know
106  // that the heat source will be small.
107  auto const xpy_area = 5. * _radius_squared - xpy_squared;
108  dealii::VectorizedArray<double> xpy_mask;
109  for (unsigned int i = 0; i < xpy_mask.size(); ++i)
110  {
111  xpy_mask[i] = xpy_area[i] < 0. ? 0. : 1.;
112  }
113  // If all the lanes of xpy_mask are zero, we can return early
114  if (xpy_mask.sum() == 0)
115  {
116  return xpy_mask;
117  }
118 
119  dealii::VectorizedArray<double> minus_three = -3.;
120  dealii::VectorizedArray<double> one = 1.;
121  dealii::VectorizedArray<double> z_rel = z / _depth;
122  dealii::VectorizedArray<double> distribution_z = z_rel * z_rel;
123  distribution_z *= minus_three;
124  distribution_z -= z_rel;
125  distribution_z -= z_rel;
126  distribution_z += one;
127 
128  dealii::VectorizedArray<double> exponent = _log_01 * xpy_squared;
129  exponent /= _radius_squared;
130  return depth_mask * _alpha * std::exp(exponent) * distribution_z;
131 }
