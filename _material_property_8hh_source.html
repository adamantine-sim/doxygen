529 template <int dim, int n_materials, int p_order, typename MaterialStates,
530  typename MemorySpaceType>
531 template <bool use_table, StateProperty state_property>
532 dealii::VectorizedArray<double>
533 MaterialProperty<dim, n_materials, p_order, MaterialStates, MemorySpaceType>::
534  compute_material_property(
535  std::array<dealii::types::material_id,
536  dealii::VectorizedArray<double>::size()> const &material_id,
537  std::array<dealii::VectorizedArray<double>,
538  MaterialStates::n_material_states> const &state_ratios,
539  dealii::VectorizedArray<double> const &temperature,
540  dealii::AlignedVector<dealii::VectorizedArray<double>> const
541  &temperature_powers) const
542 {
543  static_assert(p_order <= 4);
544  // When there is only one material, all the View indices are known at compile
545  // time. Unfortunately, even if the bounds of the for loops are known at
546  // compile time, the indices are evaluated at run time not at compile time.
547  // To get the indices evaluated at compile time, we need to unroll the loops
548  // by hand. This gives a substantial speed up for cases with few dofs.
549 
550  dealii::VectorizedArray<double> value = 0.0;
551  dealii::VectorizedArray<double> property;
552  unsigned int constexpr property_index =
553  static_cast<unsigned int>(state_property);
554 
555  if constexpr (use_table)
556  {
557  for (unsigned int material_state = 0;
558  material_state < MaterialStates::n_material_states; ++material_state)
559  {
560  for (unsigned int n = 0; n < dealii::VectorizedArray<double>::size(); ++n)
561  {
562  property[n] = compute_property_from_table(
563  _state_property_tables, material_id[n], property_index,
564  material_state, temperature[n]);
565  }
566 
567  value += state_ratios[material_state] * property;
568  }
569  }
570  else
571  {
572  if constexpr (n_materials == 1)
573  {
574  int constexpr material = 0;
575  int constexpr solid = 0;
576  int constexpr liquid = 1;
577  int constexpr powder = 2;
578 
579  // Solid
580  if constexpr (MaterialStates::n_material_states == 1)
581  {
582  // p >= 0
583  {
584  property =
585  _state_property_polynomials(material, property_index, solid, 0);
586  value += state_ratios[solid] * property * temperature_powers[0];
587  }
588 
589  // p >= 1
590  if constexpr (p_order >= 1)
591  {
592  property =
593  _state_property_polynomials(material, property_index, solid, 1);
594  value += state_ratios[solid] * property * temperature_powers[1];
595  }
596 
597  // p >= 2
598  if constexpr (p_order >= 2)
599  {
600  property =
601  _state_property_polynomials(material, property_index, solid, 2);
602  value += state_ratios[solid] * property * temperature_powers[2];
603  }
604 
605  // p >= 3
606  if constexpr (p_order >= 3)
607  {
608  property =
609  _state_property_polynomials(material, property_index, solid, 3);
610  value += state_ratios[solid] * property * temperature_powers[3];
611  }
612 
613  // p == 4
614  if constexpr (p_order == 4)
615  {
616  property =
617  _state_property_polynomials(material, property_index, solid, 4);
618  value += state_ratios[solid] * property * temperature_powers[4];
619  }
620  }
621 
622  // SolidLiquid
623  if constexpr (MaterialStates::n_material_states == 2)
624  {
625  // p >= 0
626  {
627  // Solid
628  property =
629  _state_property_polynomials(material, property_index, solid, 0);
630  value += state_ratios[solid] * property * temperature_powers[0];
631  // Liquid
632  property =
633  _state_property_polynomials(material, property_index, liquid, 0);
634  value += state_ratios[liquid] * property * temperature_powers[0];
635  }
636 
637  // p >= 1
638  if constexpr (p_order >= 1)
639  {
640  // Solid
641  property =
642  _state_property_polynomials(material, property_index, solid, 1);
643  value += state_ratios[solid] * property * temperature_powers[1];
644  // Liquid
645  property =
646  _state_property_polynomials(material, property_index, liquid, 1);
647  value += state_ratios[liquid] * property * temperature_powers[1];
648  }
649 
650  // p >= 2
651  if constexpr (p_order >= 2)
652  {
653  // Solid
654  property =
655  _state_property_polynomials(material, property_index, solid, 2);
656  value += state_ratios[solid] * property * temperature_powers[2];
657  // Liquid
658  property =
659  _state_property_polynomials(material, property_index, liquid, 2);
660  value += state_ratios[liquid] * property * temperature_powers[2];
661  }
662 
663  // p >= 3
664  if constexpr (p_order >= 3)
665  {
666  // Solid
667  property =
668  _state_property_polynomials(material, property_index, solid, 3);
669  value += state_ratios[solid] * property * temperature_powers[3];
670  // Liquid
671  property =
672  _state_property_polynomials(material, property_index, liquid, 3);
673  value += state_ratios[liquid] * property * temperature_powers[3];
674  }
675 
676  // p == 4
677  if constexpr (p_order == 4)
678  {
679  // Solid
680  property =
681  _state_property_polynomials(material, property_index, solid, 4);
682  value += state_ratios[solid] * property * temperature_powers[4];
683  // Liquid
684  property =
685  _state_property_polynomials(material, property_index, liquid, 4);
686  value += state_ratios[liquid] * property * temperature_powers[4];
687  }
688  }
689 
690  // SolidLiquidPowder
691  if constexpr (MaterialStates::n_material_states == 3)
692  {
693  // p >= 0
694  {
695  // Solid
696  property =
697  _state_property_polynomials(material, property_index, solid, 0);
698  value += state_ratios[solid] * property * temperature_powers[0];
699  // Liquid
700  property =
701  _state_property_polynomials(material, property_index, liquid, 0);
702  value += state_ratios[liquid] * property * temperature_powers[0];
703  // Powder
704  property =
705  _state_property_polynomials(material, property_index, powder, 0);
706  value += state_ratios[powder] * property * temperature_powers[0];
707  }
708 
709  // p >= 1
710  if constexpr (p_order >= 1)
711  {
712  // Solid
713  property =
714  _state_property_polynomials(material, property_index, solid, 1);
715  value += state_ratios[solid] * property * temperature_powers[1];
716  // Liquid
717  property =
718  _state_property_polynomials(material, property_index, liquid, 1);
719  value += state_ratios[liquid] * property * temperature_powers[1];
720  // Powder
721  property =
722  _state_property_polynomials(material, property_index, powder, 1);
723  value += state_ratios[powder] * property * temperature_powers[1];
724  }
725 
726  // p >= 2
727  if constexpr (p_order >= 2)
728  {
729  // Solid
730  property =
731  _state_property_polynomials(material, property_index, solid, 2);
732  value += state_ratios[solid] * property * temperature_powers[2];
733  // Liquid
734  property =
735  _state_property_polynomials(material, property_index, liquid, 2);
736  value += state_ratios[liquid] * property * temperature_powers[2];
737  // Powder
738  property =
739  _state_property_polynomials(material, property_index, powder, 2);
740  value += state_ratios[powder] * property * temperature_powers[2];
741  }
742 
743  // p >= 3
744  if constexpr (p_order >= 3)
745  {
746  // Solid
747  property =
748  _state_property_polynomials(material, property_index, solid, 3);
749  value += state_ratios[solid] * property * temperature_powers[3];
750  // Liquid
751  property =
752  _state_property_polynomials(material, property_index, liquid, 3);
753  value += state_ratios[liquid] * property * temperature_powers[3];
754  // Powder
755  property =
756  _state_property_polynomials(material, property_index, powder, 3);
757  value += state_ratios[powder] * property * temperature_powers[3];
758  }
759 
760  // p == 4
761  if constexpr (p_order == 4)
762  {
763  // Solid
764  property =
765  _state_property_polynomials(material, property_index, solid, 4);
766  value += state_ratios[solid] * property * temperature_powers[4];
767  // Liquid
768  property =
769  _state_property_polynomials(material, property_index, liquid, 4);
770  value += state_ratios[liquid] * property * temperature_powers[4];
771  // Powder
772  property =
773  _state_property_polynomials(material, property_index, powder, 4);
774  value += state_ratios[powder] * property * temperature_powers[4];
775  }
776  }
777  }
778  else
779  {
780  for (unsigned int material_state = 0;
781  material_state < MaterialStates::n_material_states; ++material_state)
782  {
783  for (unsigned int i = 0; i <= p_order; ++i)
784  {
785  for (unsigned int n = 0; n < dealii::VectorizedArray<double>::size();
786  ++n)
787  {
788  property[n] = _state_property_polynomials(
789  material_id[n], property_index, material_state, i);
790  }
791 
792  value +=
793  state_ratios[material_state] * property * temperature_powers[i];
794  }
795  }
796  }
797  }
798 
799  return value;
800 }

803  typename MemorySpaceType>
804 template <bool use_table>
805 KOKKOS_FUNCTION double
806 MaterialProperty<dim, n_materials, p_order, MaterialStates, MemorySpaceType>::
807  compute_material_property(StateProperty state_property,
808  dealii::types::material_id const material_id,
809  double const *state_ratios,
810  double temperature) const
811 {
812  double value = 0.0;
813  unsigned int const property_index = static_cast<unsigned int>(state_property);
814 
815  if constexpr (use_table)
