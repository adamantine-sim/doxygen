83 template <int dim>
84 template <int p_order, typename MaterialStates, typename MemorySpaceType>
85 void Microstructure<dim>::compute_G_and_R(
86  MaterialProperty<dim, p_order, MaterialStates, MemorySpaceType> const
87  &material_properties,
88  dealii::DoFHandler<dim> const &dof_handler,
89  dealii::LA::distributed::Vector<double, dealii::MemorySpace::Host> const
90  &temperature,
91  double delta_t)
92 {
93  // In general, the quadrature points and the support points are different.
94  // Here, we just care about G, the cooling rate, and R at some points. We
95  // don't really care which points exactly. So we can use a different
96  // quadrature that matches the support points. The big advantage is that we
97  // can easily compute which dof crossed the liquidus and compute the
98  // temperature gradient only if necessary.
99  // We are interested in the following quantities:
100  // - Temperature Gradient: G (K/m) = sqrt(G_x^2+G_y^2+G_z^2)
101  // - Cooling Rate (K/s) = |dT/dt|
102  // - Interface Velocity: R (m/s) = cooling rate / G
103 
104  // We are going to evaluate the temperature fields, so we need to update the
105  // ghost values.
106  _old_temperature.update_ghost_values();
107  temperature.update_ghost_values();
108 
109  // Create a quadrature formula that matches the support points of the finite
110  // element.
111  auto const &fe = dof_handler.get_fe();
112  dealii::QGaussLobatto<dim> quadrature_formula(fe.degree + 1);
113  dealii::FEValues<dim> fe_values(fe, quadrature_formula,
114  dealii::update_values |
115  dealii::update_gradients |
116  dealii::update_quadrature_points);
117  unsigned int const n_dofs_per_cell = fe.n_dofs_per_cell();
118  std::vector<dealii::types::global_dof_index> local_dof_indices(
119  n_dofs_per_cell);
120  std::vector<bool> relevant_indices(n_dofs_per_cell);
121  std::vector<double> G(n_dofs_per_cell);
122  std::vector<double> cooling_rate(n_dofs_per_cell);
123  std::vector<unsigned int> quad_to_support_pts(n_dofs_per_cell);
124 
125  for (const auto &cell : dof_handler.active_cell_iterators() |
126  dealii::IteratorFilters::LocallyOwnedCell() |
127  dealii::IteratorFilters::ActiveFEIndexEqualTo(0))
128  {
129  cell->get_dof_indices(local_dof_indices);
130 
131  double const liquidus =
132  material_properties.get(cell->material_id(), Property::liquidus);
133 
134  bool skip_cell = true;
135  for (unsigned int const i : fe_values.dof_indices())
136  {
137  dealii::types::global_dof_index const dof = local_dof_indices[i];
138  // Compute the cooling rate if the cell solidifies
139  if ((_old_temperature[dof] > liquidus) && (temperature[dof] <= liquidus))
140  {
141  relevant_indices[i] = true;
142  cooling_rate[i] = _old_temperature[dof] - temperature[dof] / delta_t;
143  skip_cell = false;
144  }
145  else
146  {
147  relevant_indices[i] = false;
148  }
149  }
150 
151  // The cell did not solidified, there is nothing to do.
152  if (skip_cell)
153  {
154  continue;
155  }
156 
157  // Compute G
158  fe_values.reinit(cell);
159  for (unsigned int const q_point : fe_values.quadrature_point_indices())
160  {
161  G[q_point] = 0;
162 
163  // Compute the components of the gradient
164  std::array<double, dim> grad = {0.};
165  for (unsigned int const i : fe_values.dof_indices())
166  {
167  dealii::types::global_dof_index const dof = local_dof_indices[i];
168  for (int d = 0; d < dim; ++d)
169  {
170  grad[d] += temperature[dof] * fe_values.shape_grad(i, q_point)[d];
171  }
172  // The order of the quadrature points and the support points is
173  // different. This is problematic because the cooling rate is using a
174  // different ordering than G. So we need to create a map. We use the
175  // fact that at one support point, the value of the shape function is
176  // one. At the others the value is zero.
177  if (fe_values.shape_value(i, q_point) - 1.0 > -1e-12)
178  {
179  quad_to_support_pts[q_point] = i;
180  }
181  }
182 
183  // Compute G
184  for (int d = 0; d < dim; ++d)
185  {
186  G[q_point] += std::pow(grad[d], 2);
187  }
188  G[q_point] = std::sqrt(G[q_point]);
189  }
190 
191  // Write the data on file.
192  for (unsigned int const q_point : fe_values.quadrature_point_indices())
193  {
194  unsigned int fe_index = quad_to_support_pts[q_point];
195  if (relevant_indices[fe_index])
196  {
197  for (int d = 0; d < dim; ++d)
198  {
199  _file << fe_values.quadrature_point(q_point)[d] << " ";
200  }
201  _file << G[q_point] << " " << cooling_rate[fe_index] << " "
202  << cooling_rate[fe_index] / G[q_point] << "\n";
203  }
204  }
205  }
206 }
